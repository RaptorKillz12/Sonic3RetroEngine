cmake_minimum_required(VERSION 3.24)
project(Sonic3RetroEngine VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(SDL_SHARED ON CACHE BOOL "" FORCE)
set(SDL_STATIC OFF CACHE BOOL "" FORCE)
set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3
    GIT_REPOSITORY https://github.com/libsdl-org/SDL.git
    GIT_TAG release-3.2.10
    GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL3)
add_executable(Sonic3RetroEngine src/main.cpp src/engine/Engine.cpp)
target_include_directories(Sonic3RetroEngine PRIVATE src)
target_link_libraries(Sonic3RetroEngine PRIVATE SDL3::SDL3)
if(MSVC)
    target_compile_options(Sonic3RetroEngine PRIVATE /W4 /permissive-)
else()
    target_compile_options(Sonic3RetroEngine PRIVATE -Wall -Wextra -Wpedantic)
endif()
